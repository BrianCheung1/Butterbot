def config(p):

def seed(db):
 db.execute('PRAGMA foreign_keys=ON')
 for n in (1,2):
  db.execute("INSERT INTO players VALUES (?, ?, 1, 'active')",(f'{n:032x}',n))
  db.execute("INSERT INTO economy_accounts VALUES (?, ?, 'wallet', 'coin', NULL, 1)",(f'{n+10:032x}',f'{n:032x}'))
  db.execute("INSERT INTO economy_account_balances VALUES (?, 'wallet', 25, 0)",(f'{n+10:032x}',))
 db.commit()

async def checks(p):
 result=[]
 for check in (create_database_runtime,_verify_database):
  try:
   r=await check(p)
   if hasattr(r,'close'):await r.close()
   result.append('accepted')
  except Exception as e:result.append(type(e).__name__+':'+getattr(e,'category',str(e)))
 return result
